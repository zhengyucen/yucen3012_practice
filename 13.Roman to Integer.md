#### 13.Roman to Integer
Example 1:

Input: s = "III"<br/>
Output: 3<br/>
Explanation: III = 3.<br/>

## 基本map解
```{r}
class Solution {
public:
    int romanToInt(string s) {
        map <char , int> roman{
            {'I',1}, {'V' ,5}, {'X' ,10}, {'L' ,50},
            {'C' ,100}, {'D' ,500}, {'M' ,1000}
        };
        int sum = 0;
        for(int i = 0;i < size(s) ; i++){
            if(i < size(s)-1 && roman[s[i]] < roman[s[i+1]])
                sum -= roman[s[i]];
            else
                sum += roman[s[i]];
        }
        return sum;
    }
};
```
## 更快解(map改switch)
```{r}
class Solution {
public:
    int value(char c){
            switch(c) {
                case 'I': return 1;
                case 'V': return 5;
                case 'X': return 10;
                case 'L': return 50;
                case 'C': return 100;
                case 'D': return 500;
                case 'M': return 1000;
                default: return 0;
            }
        }
    int romanToInt(string s) {
        int sum = 0;
        for(int i = 0;i < size(s) ; i++){
            if(i < size(s)-1 && value(s[i]) < value(s[i+1]))
                sum -= value(s[i]);
            else
                sum += value(s[i]);
        }
        return sum;
    }
};
```
